# Plus-end elongation at a defined velocity.
#
#   plus_end              <back|front>  which end is the barbed end (default back)
#   elongation_rate       <velocity>    plus-end growth speed, length/time (0 = off)
#   elongation_max_length <length>      stop growing past this contour length (0 = unbounded)
#
# elongation_rate is a velocity, not an event rate, so growth speed is independent of
# standard_length. Contour length grows as L(t) = L(0) + 0.32*t.
#
# smoldyn elongation2D.txt -t

graphics opengl_good

dim 2
boundaries x -0.1 0.8 r
boundaries y -0.4 0.4 r

time_start 0
time_stop 2
time_step 0.001
random_seed 1

frame_thickness 0

output_files elongation.txt

start_filament_type actin
color blue
thickness 2
polygon edge
dynamics none
kT 0.001
standard_length 0.005       # 5 nm segments
standard_angle 0            # 2D: single yaw value
force_length 500
force_angle 5               # 2D: single value
mobility 1
plus_end back               # the default; stated here for clarity
elongation_rate 0.32        # um/s at 10 uM G-actin
end_filament_type

# one filament, seeded as a single segment at the origin pointing along +x
random_filament actin:fil0 1  0 0  0

# sample the growing filament; each line is time-tagged
cmd i 0 2 0.1 printFilaments elongation.txt

end_file
