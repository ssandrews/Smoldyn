# Filament networks with loops.
#
# Either end of a filament can be free or attached to any node of any filament,
# including its own, using the "attach end target node" statement. This file
# builds four small networks that each contain a loop:
#   (1) loop:   a branch whose minus end attaches to its mother at node 2 and whose
#               plus end attaches back to the same mother at node 6
#   (2) bridge: a filament of another type (crosslink) that connects two actin
#               filaments, by its minus end to one and its plus end to the other
#   (3) lasso:  a filament whose plus end attaches to its own node 3
#   (4) circle: a filament whose plus end attaches to its own node 0
#
# Smoldyn holds attachments that form a tree exactly, by moving each attached
# filament so that its end stays on its node. Each remaining attachment closes a
# loop and is held by a spring between the end node and the target node
# (junction_force; the default is force_length). The diagnostics output at the
# start lists which attachments are rigid and which are springs. Attachments at
# interior nodes hold the attached filament at branch_angle, with stiffness
# branch_force_angle; attachments at end nodes are end-to-end joins that bend
# like any other node, with standard_angle and force_angle.
#
# smoldyn filamentLoops2D.txt

graphics opengl_good
graphic_iter 20

dim 2
boundaries x -20 20
boundaries y -20 20

time_start 0
time_stop 100
time_step 0.001
random_seed 1

frame_thickness 0

start_filament_type actin
color red
thickness 3
polygon ve
dynamics euler
kT 0.01
standard_length 1
standard_angle 0
force_length 100
force_angle 10
mobility 1
branch_angle 1.2
branch_force_angle 5
end_filament_type

start_filament_type crosslink
color green
thickness 3
polygon ve
dynamics euler
kT 0.01
standard_length 1
standard_angle 0
force_length 100
force_angle 10
mobility 1
branch_angle 1.5708          # crosslinks leave both actin filaments at right angles
branch_force_angle 5
end_filament_type

# (1) loop: a branch with both ends on its mother
start_filament actin:mother
first_segment -15 8 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
branch daughter 2 minus 1 1.2
end_filament

start_filament actin:daughter
add_segment 1 -0.6
add_segment 1 -0.6
attach plus actin:mother 6
end_filament

# (2) bridge: a crosslink between two actin filaments
start_filament actin:lower
first_segment 4 6 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
end_filament

start_filament actin:upper
first_segment 4 10 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 0
end_filament

start_filament crosslink:bridge
first_segment 8 6 1 1.5708
add_segment 1 0
add_segment 1 0
add_segment 1 0
attach minus actin:lower 4
attach plus actin:upper 4
end_filament

# (3) lasso: the plus end attaches to the filament's own node 3
start_filament actin:lasso
first_segment -16 -8 1 0
add_segment 1 0
add_segment 1 0
add_segment 1 1.0
add_segment 1 1.2566
add_segment 1 1.2566
add_segment 1 1.2566
add_segment 1 1.2566
attach plus actin:lasso 3
end_filament

# (4) circle: the plus end attaches to the filament's own minus end node
start_filament actin:circle
first_segment 8 -12 1 0
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
add_segment 1 0.5236
attach plus actin:circle 0
end_filament

cmd b diagnostics filament
cmd @ 99.999 printFilaments stdout

end_file
