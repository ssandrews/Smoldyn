# Branched filament

graphics opengl_good

dim 2
boundaries x 0 100 r
boundaries y 0 100 r

time_start 0
time_stop 100
time_step 0.01



start_filament_type mytype
color blue
thickness 2
polygon ve
dynamics none
kT 1
standard_length 2
standard_angle 0            # 2D: single yaw value
force_length 10
force_angle 5               # 2D: single value
branch_angle 1.22          # ~70 degrees, in radians
branch_spread 0.09         # ~5 degrees of angular noise
mobility 1
end_filament_type

# mother filament (10 segments): random_filament type:name segs x y phi
random_filament mytype:fil0 10  50 50  0

filament mytype:fil0 branch br0 4 minus 3 -1
filament mytype:br0 add_segment 1 0

filament mytype:fil0 branch br1 5 plus 3 0.2
filament mytype:br1 add_segment 1 -0.2 2 minus

end_file
