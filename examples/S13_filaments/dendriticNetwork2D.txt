# A growing branched-actin network: branching + elongation + capping.
#
# Deliberately no treadmill_rate. Treadmilling forces equal flux at both ends, while
# pointed-end kinetics are ~9x slower and Arp2/3 caps the daughter's pointed end, so
# elongation is the right growth mechanism for a dendritic network.
#
# For a time-lapse, swap the cmd line below for:
#   cmd i 0 3 0.05 printFilaments dendriticNetwork.txt
#
# smoldyn dendriticNetwork2D.txt -t

graphics opengl_good

dim 2
boundaries x -1 1 r
boundaries y -1 1 r

time_start 0
time_stop 3
time_step 0.001
random_seed 1

frame_thickness 0

output_files dendriticNetwork.txt

start_filament_type actin
color blue
thickness 2
polygon edge
dynamics none
kT 0.001
standard_length 0.005       # 5 nm segments
standard_angle 0            # 2D: single yaw value
force_length 500
force_angle 5               # 2D: single value
mobility 1

elongation_rate 0.32        # um/s barbed-end growth
capping_rate 4.3            # /s -> mean filament length 74 nm

branch_rate 40              # Arp2/3 nucleation per unit mother length per time
branch_angle 1.22           # ~70 degrees, in radians
branch_spread 0.09          # ~5 degrees of angular noise
end_filament_type

# three seed filaments; every other filament in the run is a nucleated daughter
random_filament actin:seed0 1   0.0  0.0   0
random_filament actin:seed1 1  -0.2  0.1   1
random_filament actin:seed2 1   0.2 -0.1  -1

# dump the whole network (all auto-named daughters included) at the end
cmd @ 3 printFilaments dendriticNetwork.txt

end_file
